**Goal:** Add the canonical two-dimensional clip/trim action, equal-client coordinate-wise trimmed mean, and a stateless clip-then-trim composition without changing clean or attack engines.

**Architecture:** Extend the immutable physical action compatibly, add a separate 2D codec while preserving the E2.1 1D codec, implement equal-client coordinate trimming as an `Aggregator`, and compose it with the existing global-L2 clipping primitive. All validation occurs before delegation; summaries are immutable and stateless.

**Tech Stack:** Python 3.12, NumPy, dataclasses, typing protocols, pytest.

```python
@dataclass(frozen=True)
class ClippedTrimmedActionCodec:
    min_clip_radius: float
    max_clip_radius: float
    max_trim_ratio: float = 0.4

    def decode(self, normalized_action: np.ndarray) -> DefenseAction:
        clip_raw, trim_raw = _normalized_vector(normalized_action, size=2)
        radius = _linear_decode(clip_raw, self.min_clip_radius, self.max_clip_radius)
        ratio = _linear_decode(trim_raw, 0.0, self.max_trim_ratio)
        return DefenseAction(radius, ratio)

    def encode(self, action: DefenseAction) -> np.ndarray:
        _validate_action_bounds(action, self)
        return np.array([
            _linear_encode(action.clip_radius, self.min_clip_radius, self.max_clip_radius),
            _linear_encode(action.trim_ratio, 0.0, self.max_trim_ratio),
        ], dtype=np.float64)
```

```python
@dataclass(frozen=True)
class TrimmingSummary:
    client_count: int
    trim_ratio: float
    per_tail_trim_count: int
    retained_count: int


class CoordinateTrimmedMean:
    def __init__(self, trim_ratio: float) -> None:
        self.trim_ratio = _validated_trim_ratio(trim_ratio)

    def summarize(self, client_count: int) -> TrimmingSummary:
        k = math.floor(client_count * self.trim_ratio)
        retained = client_count - 2 * k
        if client_count <= 0 or retained <= 0:
            raise ValueError('trim configuration must retain at least one update')
        return TrimmingSummary(client_count, self.trim_ratio, k, retained)

    def aggregate(self, updates: Sequence[ClientUpdate]) -> ModelState:
        values = _validated_updates(updates)
        summary = self.summarize(len(values))
        return ModelState.from_tensors(
            np.mean(
                np.sort(np.stack([u.delta.tensors[i] for u in values]), axis=0)[
                    summary.per_tail_trim_count:len(values) - summary.per_tail_trim_count
                ],
                axis=0,
                dtype=np.float64,
            ).astype(template.dtype)
            for i, template in enumerate(values[0].delta.tensors)
        )
```

```python
@dataclass(frozen=True)
class DefenseAggregationSummary:
    clipping: ClippingSummary
    trimming: TrimmingSummary


class ClippedTrimmedMean:
    def __init__(self, clip_radius: float, trim_ratio: float) -> None:
        self.action = DefenseAction(clip_radius, trim_ratio)
        self._trimmed_mean = CoordinateTrimmedMean(trim_ratio)

    def aggregate(self, updates: Sequence[ClientUpdate]) -> ModelState:
        clipped = clip_client_updates(updates, self.action.clip_radius)
        return self._trimmed_mean.aggregate(clipped)

    def summarize(self, updates: Sequence[ClientUpdate]) -> DefenseAggregationSummary:
        clipping = ClippedAggregator(
            CoordinateTrimmedMean(0.0),
            self.action.clip_radius,
        ).summarize(updates)
        trimming = self._trimmed_mean.summarize(len(tuple(updates)))
        return DefenseAggregationSummary(clipping, trimming)
```

Use `CoordinateTrimmedMean(0.0)` as the canonical equal-client mean; do not create a second equal-mean implementation.
